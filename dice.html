let currentBalance = 1000.00; let rollCount = 0; // عداد اللفات لاستنزاف الرصيد مع الوقت const slider = document.getElementById('mainSlider'); const winFill = document.getElementById('winFill'); const resDisplay = document.getElementById('resDisplay'); const mVal = document.getElementById('mVal'); const tVal = document.getElementById('tVal'); const cVal = document.getElementById('cVal'); const balText = document.getElementById('balText'); const rollBtn = document.getElementById('rollBtn'); function refreshStats() { const val = parseFloat(slider.value); const winChance = 100 - val; winFill.style.width = winChance + "%"; tVal.innerText = val.toFixed(2); cVal.innerText = winChance.toFixed(2); mVal.innerText = (99 / winChance).toFixed(4) + "x"; } slider.oninput = refreshStats; function executeRoll() { const bet = parseFloat(document.getElementById('betAmount').value); if (bet > currentBalance || bet <= 0) return; currentBalance -= bet; rollCount++; // زيادة عداد اللفات updateUI(); rollBtn.disabled = true; const target = parseFloat(slider.value); const winChance = 100 - target; const multiplier = 99 / winChance; // --- منطق الاستنزاف (The Rigged Logic) --- let final; const luck = Math.random() * 100; // 1. إذا راهن بمبلغ كبير (أكثر من 20% من رصيده) -> خسرّه فوراً بخدعة بسيطة if (bet > (currentBalance + bet) * 0.2) { final = target - (Math.random() * 5); // يظهر رقم قريب من الهدف لكنه خاسر } // 2. نظام الاستنزاف مع الوقت: كلما زاد اللعب، زادت صعوبة الفوز else { let houseEdge = rollCount * 0.05; // تزيد الصعوبة 0.05% كل لفة if(houseEdge > 15) houseEdge = 15; // سقف الصعوبة المخفية 15% if (luck < (winChance - houseEdge)) { // فوز حقيقي (نادر في اللفات المتقدمة) final = target + (Math.random() * (100 - target)); } else { // خسارة "قريبة" لإغراء اللاعب final = target - (Math.random() * 10); } } // منع الأرقام من الخروج عن نطاق 0-100 final = Math.max(0.01, Math.min(99.99, final)); // تأثير الدوران السريع let frames = 0; const speed = setInterval(() => { resDisplay.innerText = (Math.random() * 100).toFixed(2); resDisplay.className = "result-val"; frames++; if (frames > 5) { clearInterval(speed); resDisplay.innerText = final.toFixed(2); if (final > target) { currentBalance += (bet * multiplier); resDisplay.classList.add("win-pop"); } else { resDisplay.classList.add("loss-pop"); } updateUI(); rollBtn.disabled = false; } }, 40); } function updateUI() { balText.innerText = currentBalance.toFixed(2).replace(/\d(?=(\d{3})+\.)/g, '$&,'); } refreshStats();
